cmake_minimum_required(VERSION 3.10)
project(tmp)
add_compile_options("$<$<C_COMPILER_ID:MSVC>:/utf-8>")
add_compile_options("$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
find_package(OpenCV REQUIRED)
add_executable(a 1.cpp)
add_executable(b 2.cpp)
target_link_libraries(a ${OpenCV_LIBS})
target_link_libraries(b ${OpenCV_LIBS})
target_include_directories(a PRIVATE ${OpenCV_INCLUDE_DIRS})
target_include_directories(b PRIVATE ${OpenCV_INCLUDE_DIRS})
